  list(APPEND CCORD_RPC_SOURCES game_scan_win32.c)

  list(APPEND CCORD_RPC_SOURCES game_scan_macos.c)

  list(APPEND CCORD_RPC_SOURCES game_scan_linux.c)

endif()
