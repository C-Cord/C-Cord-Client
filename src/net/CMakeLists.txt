    target_link_libraries(ccord_net PUBLIC PkgConfig::CCORD_SPEEXDSP)
  endif()
endif()
if(CCORD_LIVE_ACCOUNT AND CCORD_SPEEXDSP_FOUND)
  target_compile_definitions(ccord_net PUBLIC CCORD_HAVE_SPEEXDSP=1)
else()
  target_compile_definitions(ccord_net PUBLIC CCORD_HAVE_SPEEXDSP=0)
endif()
if(CCORD_LIVE_ACCOUNT)
  target_compile_definitions(ccord_net PRIVATE CCORD_PROFILE_LIVE=1)
endif()
